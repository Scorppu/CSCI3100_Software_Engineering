@startuml
skinparam classAttributeIconSize 0

interface Antigen {
// Substance that can be used to identify
// whether the foreigner is an enemy,
// and the kind of enemy
}

class Enemy {
+has_been_seen_before : Boolean
+antigens : [Antigen]
}

class Antibody {
-enemy : Enemy
+set_enemy(enemy: Enemy)
+can_kill_enemy(enemy: Enemy) -> Boolean
+kill_enemy(enemy: Enemy) -> Boolean
}

interface WhiteBloodCell {
+is_enemy(antigens: [Antigen]) -> Boolean
+kill(enemy: Enemy) -> Boolean
}

class Neutrophils {
+kill(enemy: Enemy) -> Boolean
+is_enemy(antigens: [Antigen]) -> Boolean
}

class Lymphocytes {
+is_enemy(antigens: [Antigen]) -> Boolean
+kill(enemy: Enemy) -> Boolean
}

class NKCell {
+is_enemy(antigens: [Antigen]) -> Boolean
+kill(enemy: Enemy) -> Boolean
}

class BCell {
+is_enemy(antigens: [Antigen]) -> Boolean
+kill(enemy: Enemy) -> Boolean
+make_antibodies(enemy: Enemy) -> Antibody
+ask_helper_t_cell_make_antibodies(helper_t_cell: HelperTCell)
}

class TCell {
+is_activated : Boolean
+is_enemy(antigens: [Antigen]) -> Boolean
+kill(enemy: Enemy) -> Boolean
}

class HelperTCell {
+get_enemy_characteristics(enemy: Enemy) -> [float]
}


Neutrophils ..|> WhiteBloodCell
Neutrophils ..> Enemy
WhiteBloodCell ..> Antigen

Lymphocytes ..|> WhiteBloodCell
Lymphocytes ..> Enemy

Lymphocytes --* NKCell
Lymphocytes --* TCell
Lymphocytes --* BCell 
BCell ..> Antibody
BCell ..> HelperTCell

Antibody --> Enemy
HelperTCell *-- TCell

Enemy --* Antigen
@enduml
